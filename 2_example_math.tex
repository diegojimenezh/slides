\subsection{Including math}

\begin{frame}{Examples of display-style and inline math}

\begin{itemize}

\item \textbf{Display-style math:} To illustrate display-style math, consider the Wald estimator:
\[
\tau^{IV} = \frac{\mathbb{E}[Y_i | Z_i = 1] - \mathbb{E}[Y_i | Z_i = 0]}{\mathbb{E}[D_i | Z_i = 1] - \mathbb{E}[D_i | Z_i = 0]}
\]

\vfill 

\item \textbf{Inline math:} For example, a simple linear regression model can be written inline as $Y = X' \beta + \varepsilon$. 

\end{itemize}

\end{frame}
