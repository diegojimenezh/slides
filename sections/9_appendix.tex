% Prints a color's hex code, read from its \definecolor.
\newcommand{\hexof}[1]{%
	\extractcolorspecs{#1}{\hexmodel}{\hexspec}%
	\convertcolorspec{\hexmodel}{\hexspec}{HTML}{\hexval}%
	\#\hexval}

\begin{tikzpicture}[
	swatch/.style = {rectangle, minimum width = 1.55cm, minimum height = 0.7cm,
	                 draw = rule, line width = 0.4pt},

		accent/accent, accentalt/accentalt, highlight/highlight}

		\node[swatch, fill = \c] (s\i) at (\i*1.65, 0.65) {};

	}
